#include <stdio.h>
#include <unistd.h>
#include <signal.h>

void handler(int sig)
{
    if (sig == SIGINT)
        printf("\nSIGINT received\n");

    if (sig == SIGTERM)
        printf("\nSIGTERM received\n");

    if (sig == SIGUSR1)
        printf("\nSIGUSR1 received\n");
}

int main()
{
    signal(SIGINT, handler);
    signal(SIGTERM, handler);
    signal(SIGUSR1, handler);

    printf("Process ID: %d\n", getpid());

    while (1)
    {
        sleep(1);
    }

    return 0;
}
